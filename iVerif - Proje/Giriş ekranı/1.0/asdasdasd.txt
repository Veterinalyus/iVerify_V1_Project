<script>


let currentData = null;

let currentCategory = "mutual";

let currentUsers = [];





const API =
"http://127.0.0.1:5000";





// =====================================================
// ZIP UPLOAD
// =====================================================



document
.getElementById("file-input")
.addEventListener(
"change",
function(e){


    if(
        e.target.files.length
    ){

        uploadArchive(
            e.target.files[0]
        );

    }


});







async function uploadArchive(file){



    let form =
    new FormData();



    form.append(
        "instagram_archive",
        file
    );



    document
    .getElementById("loading")
    .classList
    .remove("hidden");



    try{


        let response =
        await fetch(

            API +
            "/api/arsivle",

            {

                method:"POST",

                body:form

            }

        );



        let json =
        await response.json();



        if(
            json.success
        ){


            currentData =
            json.data;



            showResults(
                currentData
            );


        }

        else{


            alert(
                json.error
            );

        }



    }

    catch(error){


        alert(
            "Sunucu bağlantısı kurulamadı."
        );


    }



    finally{


        document
        .getElementById("loading")
        .classList
        .add("hidden");


    }


}









// =====================================================
// SHOW RESULTS
// =====================================================



function showResults(data){



    document
    .getElementById("results")
    .classList
    .remove("hidden");




    document
    .getElementById("followers")
    .innerText =
    data.stats.followers;



    document
    .getElementById("following")
    .innerText =
    data.stats.following;



    document
    .getElementById("mutual")
    .innerText =
    data.stats.mutual;



    document
    .getElementById("pending")
    .innerText =
    data.stats.pending_requests;




    changeCategory(
        "mutual"
    );


}









// =====================================================
// TAB SYSTEM
// =====================================================



document
.querySelectorAll(".tab")
.forEach(

function(button){



button.addEventListener(

"click",

function(){



document
.querySelectorAll(".tab")
.forEach(

b =>
b.classList.remove(
"active"
)

);



this.classList.add(
"active"
);




changeCategory(
this.dataset.tab
);



}



);



}

);









function changeCategory(category){



currentCategory =
category;



if(
!currentData
||
!currentData.users
)
return;



currentUsers =
currentData
.users
[category]
||
[];




renderUsers();

}









// =====================================================
// USER LIST
// =====================================================



function renderUsers(){



let container =
document
.getElementById(
"user-list"
);



container.innerHTML="";



let search =
document
.getElementById(
"search"
)
.value
.toLowerCase();




let filtered =

currentUsers.filter(

user =>

user.username
.toLowerCase()
.includes(search)

);





filtered.forEach(

function(user){



let row =
document.createElement(
"div"
);



row.className =
"user-item";




row.innerHTML = `


<span
class="user-name"
onclick="showDetail('${user.username}')"
>

@${user.username}

</span>



<a

class="profile-btn"

target="_blank"

href="${user.instagram}"

>

Instagram

</a>



`;




container.appendChild(
row
);



}

);



}









// =====================================================
// SEARCH
// =====================================================



document
.getElementById("search")
.addEventListener(

"input",

function(){

renderUsers();

}

);









// =====================================================
// USER DETAIL
// =====================================================



function showDetail(username){



let user =

currentUsers.find(

u =>
u.username === username

);



if(!user)
return;




alert(

"@" +
user.username +

"\n\n" +

"Kaynak:\n" +

(user.source || "Bilinmiyor")

+

"\n\nTarih:\n"

+

(

user.timestamp
?
new Date(
user.timestamp * 1000
)
.toLocaleString()

:

"Veride bulunamadı"

)

);



}









// =====================================================
// FUTURE COMPARISON READY
// =====================================================



async function compareArchives(
oldId,
newId
){



try{


let response =

await fetch(

API +

"/api/karsilastir/"

+

oldId

+

"/"

+

newId

);



return await response.json();



}

catch(e){


console.log(e);


}



}

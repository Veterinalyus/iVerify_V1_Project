def get_db():

    connection = sqlite3.connect(
        DATABASE_FILE
    )

    connection.row_factory = sqlite3.Row

    return connection





def init_database():


    with get_db() as db:


        db.execute(
            """
            CREATE TABLE IF NOT EXISTS archives
            (

                id TEXT PRIMARY KEY,

                created_at INTEGER,

                filename TEXT,


                followers_count INTEGER,

                following_count INTEGER,


                mutual_count INTEGER,


                not_following_back_count INTEGER,


                fans_count INTEGER,


                pending_requests_count INTEGER

            )
            """
        )



        db.execute(
            """
            CREATE TABLE IF NOT EXISTS users
            (

                id INTEGER PRIMARY KEY AUTOINCREMENT,


                archive_id TEXT,


                username TEXT,


                category TEXT,


                timestamp INTEGER,


                source TEXT,


                FOREIGN KEY(
                    archive_id
                )
                REFERENCES archives(id)

            )
            """
        )



        db.execute(
            """
            CREATE TABLE IF NOT EXISTS comparisons
            (

                id INTEGER PRIMARY KEY AUTOINCREMENT,


                old_archive TEXT,


                new_archive TEXT,


                created_at INTEGER

            )
            """
        )



        db.commit()




init_database()

def success(data):

    return jsonify(

        {
            "success": True,

            "data": data,

            "error": None

        }

    )





def error(message):

    return jsonify(

        {
            "success": False,

            "data": None,

            "error": message

        }

    )

def check_rate_limit(ip):


    now = time.time()



    with request_lock:


        old = [

            x

            for x in request_cache[ip]

            if now - x < RATE_LIMIT_TIME

        ]



        if len(old) >= RATE_LIMIT_COUNT:

            return False



        old.append(now)


        request_cache[ip] = old



    return True







def clean_username(username):


    if not isinstance(
        username,
        str
    ):

        return None



    username = (

        username

        .strip()

        .replace("@","")

        .lower()

    )



    if not username:

        return None



    if len(username) > MAX_USERNAME_LENGTH:

        return None



    return username





def safe_timestamp(value):


    try:

        return int(value)


    except:

        return 0

def check_zip_path(name):


    name = name.replace(
        "\\",
        "/"
    )



    if ".." in name:

        return False



    if name.startswith("/"):

        return False



    return True





def validate_zip(path):


    with zipfile.ZipFile(
        path,
        "r"
    ) as archive:


        files = archive.infolist()



        if len(files) > MAX_ZIP_FILES:

            raise Exception(
                "ZIP içinde fazla dosya var."
            )



        for item in files:


            if not check_zip_path(
                item.filename
            ):

                raise Exception(
                    "Güvenli olmayan ZIP yolu."
                )



            if item.file_size > MAX_SINGLE_FILE_SIZE:

                raise Exception(
                    "Dosya boyutu çok büyük."
                )



    return True





# ============================================================
# JSON DEPTH CONTROL
# ============================================================



def check_json_depth(
        data,
        depth=0
):


    if depth > MAX_JSON_DEPTH:

        raise Exception(
            "JSON yapısı çok derin."
        )



    if isinstance(
        data,
        dict
    ):


        for value in data.values():

            check_json_depth(
                value,
                depth + 1
            )



    elif isinstance(
        data,
        list
    ):


        for item in data:

            check_json_depth(
                item,
                depth + 1
            )

# ------------------------------------------------------------


def extract_username(item):


    if not isinstance(item, dict):

        return None



    # Yeni Meta formatı

    if "title" in item:

        username = clean_username(
            item.get("title")
        )


        if username:

            return username




    # Eski Meta formatı

    data_list = item.get(
        "string_list_data",
        []
    )



    if isinstance(
        data_list,
        list
    ):


        for data in data_list:


            if isinstance(
                data,
                dict
            ):


                username = clean_username(

                    data.get(
                        "value"
                    )

                    or

                    data.get(
                        "username"
                    )

                )



                if username:

                    return username



    return None





def extract_timestamp(item):


    if not isinstance(
        item,
        dict
    ):

        return 0



    data_list = item.get(
        "string_list_data",
        []
    )



    if isinstance(
        data_list,
        list
    ):


        for data in data_list:


            if isinstance(
                data,
                dict
            ):


                return safe_timestamp(

                    data.get(
                        "timestamp",
                        0
                    )

                )



    return 0







# ------------------------------------------------------------
# JSON TYPE DETECTION
# ------------------------------------------------------------


def detect_file_type(
        filename,
        data
):


    name = filename.lower()



    # Bekleyen istekler

    if (

        "request" in name

        or

        "pending" in name

    ):

        return "pending"




    # Following


    if (

        "following" in name

        or

        "relationships_following" in str(data)

    ):

        return "following"





    # Followers


    if "follower" in name:

        return "followers"




    return None







# ------------------------------------------------------------
# DATA LIST FINDER
# ------------------------------------------------------------


def find_list_data(data):


    if isinstance(
        data,
        list
    ):

        return data



    if isinstance(
        data,
        dict
    ):



        for key,value in data.items():


            if isinstance(
                value,
                list
            ):

                return value



    return []

def analyze_archive(zip_path):


    followers = {}

    following = {}

    pending_requests = {}



    checked_files = []

    skipped_files = []




    with zipfile.ZipFile(
        zip_path,
        "r"
    ) as archive:



        for file in archive.infolist():



            if file.is_dir():

                continue



            if not file.filename.lower().endswith(
                ".json"
            ):

                continue



            checked_files.append(
                file.filename
            )



            try:



                with archive.open(
                    file
                ) as f:


                    data = json.load(
                        f
                    )



                check_json_depth(
                    data
                )



                file_type = detect_file_type(

                    file.filename,

                    data

                )



                items = find_list_data(
                    data
                )




                for item in items:



                    username = extract_username(
                        item
                    )



                    if not username:

                        continue



                    user_data = {


                        "username":

                            username,


                        "timestamp":

                            extract_timestamp(
                                item
                            ),



                        "source":

                            file.filename,



                        "instagram":

                            "https://instagram.com/" + username

                    }



                    if file_type == "followers":



                        followers[username] = user_data





                    elif file_type == "following":



                        following[username] = user_data





                    elif file_type == "pending":



                        pending_requests[username] = user_data





            except Exception as e:



                skipped_files.append(

                    {

                        "file":

                            file.filename,


                        "error":

                            str(e)

                    }

                )







    # --------------------------------------------------------
    # RELATION CALCULATIONS
    # --------------------------------------------------------



    followers_set = set(
        followers.keys()
    )


    following_set = set(
        following.keys()
    )



    mutual = (

        followers_set

        &

        following_set

    )



    following_not_following_back = (

        following_set

        -

        followers_set

    )



    followers_not_following = (

        followers_set

        -

        following_set

    )







    result = {



        "stats": {



            "followers":

                len(followers_set),



            "following":

                len(following_set),



            "mutual":

                len(mutual),



            "not_following_back":

                len(
                    following_not_following_back
                ),



            "fans":

                len(
                    followers_not_following
                ),



            "pending_requests":

                len(
                    pending_requests
                )

        },




        "users": {



            "mutual":

                [

                    followers[x]

                    for x in sorted(
                        mutual
                    )

                ],





            "following_not_following_back":

                [

                    following[x]

                    for x in sorted(
                        following_not_following_back
                    )

                ],





            "followers_not_following":

                [

                    followers[x]

                    for x in sorted(
                        followers_not_following
                    )

                ],





            "followers":

                list(
                    followers.values()
                ),





            "following":

                list(
                    following.values()
                ),





            "pending_requests":

                list(
                    pending_requests.values()
                )

        },




        "analysis": {



            "checked_files":

                len(
                    checked_files
                ),



            "skipped_files":

                skipped_files,


            "status":

                "completed"

        }


    }



    return result

# ------------------------------------------------------------
# SAVE ARCHIVE
# ------------------------------------------------------------


def save_archive(
        result,
        filename
):


    archive_id = uuid.uuid4().hex


    now = int(
        time.time()
    )



    with get_db() as db:



        db.execute(

            """

            INSERT INTO archives

            (

                id,

                created_at,

                filename,


                followers_count,

                following_count,


                mutual_count,


                not_following_back_count,


                fans_count,


                pending_requests_count

            )


            VALUES

            (?,?,?,?,?,?,?,?,?)

            """,

            (

                archive_id,

                now,

                filename,


                result["stats"]["followers"],

                result["stats"]["following"],

                result["stats"]["mutual"],

                result["stats"]["not_following_back"],

                result["stats"]["fans"],

                result["stats"]["pending_requests"]

            )

        )




        for category, users in result["users"].items():



            for user in users:


                db.execute(

                    """

                    INSERT INTO users

                    (

                        archive_id,

                        username,

                        category,

                        timestamp,

                        source

                    )


                    VALUES

                    (?,?,?,?,?)

                    """,

                    (

                        archive_id,

                        user["username"],

                        category,

                        user.get(
                            "timestamp",
                            0
                        ),

                        user.get(
                            "source",
                            ""
                        )

                    )

                )



        db.commit()



    return archive_id

@app.route(
    "/api/arsivle",
    methods=["POST"]
)

def upload_archive():



    ip = request.remote_addr



    if not check_rate_limit(ip):


        return error(
            "Çok fazla istek gönderildi."
        )




    if "instagram_archive" not in request.files:


        return error(
            "ZIP dosyası bulunamadı."
        )




    uploaded = request.files[
        "instagram_archive"
    ]



    filename = secure_filename(
        uploaded.filename
    )



    if not filename.lower().endswith(
        ".zip"
    ):


        return error(
            "Sadece ZIP dosyaları kabul edilir."
        )




    file_path = (

        UPLOAD_DIR /

        (

            uuid.uuid4().hex

            +

            "_"

            +

            filename

        )

    )



    try:



        uploaded.save(
            file_path
        )



        validate_zip(
            file_path
        )



        result = analyze_archive(
            file_path
        )



        archive_id = save_archive(

            result,

            filename

        )



        result["archive_id"] = archive_id



        return success(
            result
        )




    except Exception as e:



        logger.exception(
            "Analiz hatası"
        )


        return error(
            str(e)
        )



    finally:



        if file_path.exists():

            file_path.unlink()

            """

        [

            for row in rows

    )

@app.route(
    "/api/karsilastir/<old_id>/<new_id>",
    methods=["GET"]
)

def compare_archives(
        old_id,
        new_id
):


    with get_db() as db:



        old_users = db.execute(

            """

            SELECT username, category

            FROM users

            WHERE archive_id=?

            """,

            (
                old_id,
            )

        ).fetchall()




        new_users = db.execute(

            """

            SELECT username, category

            FROM users

            WHERE archive_id=?

            """,

            (
                new_id,
            )

        ).fetchall()





    old = {

        (

            x["username"],

            x["category"]

        )

        for x in old_users

    }




    new = {

        (

            x["username"],

            x["category"]

        )

        for x in new_users

    }





    return success(


        {


            "added":

                [

                    {

                        "username":x[0],

                        "category":x[1]

                    }

                    for x in sorted(
                        new-old
                    )

                ],




            "removed":

                [

                    {

                        "username":x[0],

                        "category":x[1]

                    }

                    for x in sorted(
                        old-new
                    )

                ]

        }

    )

const API_URL = "http://127.0.0.1:5000";



let analysisData = null;

// -----------------------------------------------------
// ZIP UPLOAD
// -----------------------------------------------------



document
.getElementById("zipFile")
.addEventListener(

"change",

function(e){


    if(e.target.files.length){


        uploadZIP(
            e.target.files[0]
        );


    }


}

);







async function uploadZIP(file){



    let formData = new FormData();



    formData.append(

        "file",

        file

    );




    document
    .getElementById("loading")
    .classList
    .remove("hidden");





    try{


        let response = await fetch(

            API_URL +

            "/api/analyze",

            {

                method:"POST",

                body:formData

            }

        );





        let result = await response.json();




        if(result.success){



            analysisData = result.data;



            showAnalysis(

                analysisData

            );



        }


        else{


            alert(

                result.error

            );


        }



    }


    catch(error){


        alert(

            "Python motoruna bağlanılamadı."

        );


        console.log(error);



    }



    finally{


        document
        .getElementById("loading")
        .classList
        .add("hidden");


    }



}









// -----------------------------------------------------
// SHOW ANALYSIS
// -----------------------------------------------------



function showAnalysis(data){



    document
    .getElementById("resultArea")
    .classList
    .remove("hidden");



    document
    .getElementById("usersArea")
    .classList
    .remove("hidden");






    document
    .getElementById("followersCount")
    .innerText =

    data.stats.followers;





    document
    .getElementById("followingCount")

    data.stats.following;






    document
    .getElementById("mutualCount")
    .innerText =

    data.stats.mutual;






    document
    .getElementById("pendingCount")
    .innerText =

    data.stats.pending_requests;






    changeCategory(

        "mutual"

    );



}

function changeCategory(type){



currentCategory = type;



if(!analysisData)
return;




currentUsers =

analysisData.users[type]

||
[];





renderUsers();



}









// -----------------------------------------------------
// USER LIST
// -----------------------------------------------------



function renderUsers(){



let list =

document
.getElementById("userList");



list.innerHTML = "";




let searchText =

document
.getElementById("searchBox")
.value
.toLowerCase();






let filtered =

currentUsers.filter(

user =>


user.username

.toLowerCase()

.includes(

searchText

)


);







filtered.forEach(

function(user){



let row =

document.createElement(
"div"
);



row.className =
"user-row";





row.innerHTML = `


<span

class="username"

onclick="showUser('${user.username}')"

>

@${user.username}

</span>



<a

class="profile-link"

href="${user.url}"

target="_blank"

>

Instagram

</a>



`;





list.appendChild(
row
);



}



);



}

function showUser(username){



    let user =

    currentUsers.find(

        u =>

        u.username === username

    );




    if(!user)

        return;






    document

    .getElementById(

        "detailUsername"

    )

    .innerHTML =

    "<b>Kullanıcı:</b> @" +

    user.username;







    document

    .getElementById(

        "detailStatus"

    )

    .innerHTML =

    "<b>Durum:</b> " +

    (

        user.status

        ||

        "Bilinmiyor"

    );








    let date =

    "Veride bulunamadı";





    if(user.timestamp){



        date =

        new Date(

            user.timestamp * 1000

        )

        .toLocaleString();



    }







    document

    .getElementById(

        "detailDate"

    )

    .innerHTML =

    "<b>Tarih:</b> " +

    date;








    document

    .getElementById(

        "detailSource"

    )

    .innerHTML =

    "<b>Kaynak:</b> " +

    (

        user.source

        ||

        "Bilinmiyor"

    );







    document

    .getElementById(

        "detailInstagram"

    )

    .href =

    user.url;







    document

    .getElementById(

        "userModal"

    )

    .style.display =

    "flex";





}

// =====================================================
// ARCHIVE COMPARISON READY
// =====================================================




async function compareArchives(

oldArchive,

newArchive

){



    try{



        let response =

        await fetch(

            API_URL +

            "/api/compare/" +

            oldArchive +

            "/" +

            newArchive

        );





        let result =

        await response.json();





        return result;



    }

    catch(error){



        console.log(error);



        return null;



    }



}








// =====================================================
// SORT SYSTEM READY
// =====================================================




function sortUsers(mode){



    if(!currentUsers)

        return;





    if(mode==="az"){



        currentUsers.sort(

            (a,b)=>

            a.username.localeCompare(

                b.username

            )

        );



    }





    if(mode==="za"){



        currentUsers.sort(

            (a,b)=>

            b.username.localeCompare(

                a.username

            )

        );



    }





    renderUsers();



}

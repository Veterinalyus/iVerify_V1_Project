// =====================================================
// iVerify Interface v3
// PART 2
// API + LIST ENGINE
// =====================================================



const API_URL = "http://127.0.0.1:5000";



let analysisData = null;


let currentUsers = [];


let currentPage = 1;


let pageSize = 50;


let currentCategory = "mutual";






// -----------------------------------------------------
// ZIP UPLOAD
// -----------------------------------------------------



document

.getElementById("zipFile")

.addEventListener(

"change",

function(e){


    if(e.target.files.length){


        uploadZip(

            e.target.files[0]

        );


    }


}

);







async function uploadZip(file){



    let formData = new FormData();



    formData.append(

        "file",

        file

    );





    document

    .getElementById("loading")

    .classList

    .remove("hidden");






    try{


        let response = await fetch(

            API_URL +

            "/api/analyze",

            {

                method:"POST",

                body:formData

            }

        );





        let result = await response.json();





        if(result.success){


            analysisData = result.data;



            showDashboard();



        }

        else{


            alert(

                result.error

            );


        }





    }

    catch(error){



        alert(

            "Python motoruna bağlanılamadı"

        );



        console.log(error);



    }



    finally{


        document

        .getElementById("loading")

        .classList

        .add("hidden");



    }



}









// -----------------------------------------------------
// DASHBOARD
// -----------------------------------------------------



function showDashboard(){



    document

    .getElementById("dashboard")

    .classList

    .remove("hidden");




    document

    .getElementById("users")

    .classList

    .remove("hidden");






    document

    .getElementById("followers")

    .innerText =

    analysisData.stats.followers;






    document

    .getElementById("following")

    .innerText =

    analysisData.stats.following;






    document

    .getElementById("mutual")

    .innerText =

    analysisData.stats.mutual;






    document

    .getElementById("pending")

    .innerText =

    analysisData.stats.pending;






    changeCategory(

        "mutual"

    );



}









// -----------------------------------------------------
// CATEGORY CHANGE
// -----------------------------------------------------



document

.querySelectorAll(".category")

.forEach(

button => {



button.addEventListener(

"click",

function(){



document

.querySelectorAll(".category")

.forEach(

b =>

b.classList.remove(

"active"

)

);





this.classList.add(

"active"

);





changeCategory(

this.dataset.type

);



}

);



}

);









function changeCategory(type){



currentCategory = type;



currentPage = 1;





currentUsers =

analysisData.users[type]

||

[];





renderUsers();



}









// -----------------------------------------------------
// USER LIST
// -----------------------------------------------------



function renderUsers(){



let list =

document

.getElementById("userList");



list.innerHTML = "";





let text =

document

.getElementById("search")

.value

.toLowerCase();







let users =

currentUsers.filter(

u =>



u.username

.toLowerCase()

.includes(text)



);








let start =

(currentPage-1)

*

pageSize;





let end =

start +

pageSize;






let pageUsers =

users.slice(

start,

end

);






pageUsers.forEach(

user => {



let row =

document.createElement(

"div"

);



row.className =

"user-row";





row.innerHTML = `



<span

onclick="openUser('${user.username}')"

>

@${user.username}

</span>




<a

class="profile"

href="${user.url}"

target="_blank"

>

Instagram

</a>



`;





list.appendChild(

row

);



}

);



document

.getElementById("page")

.innerText =

currentPage;



}









// -----------------------------------------------------
// SEARCH
// -----------------------------------------------------



document

.getElementById("search")

.addEventListener(

"input",

function(){


currentPage = 1;


renderUsers();



}

);









// -----------------------------------------------------
// PAGINATION
// -----------------------------------------------------



document

.getElementById("next")

.onclick = function(){



currentPage++;



renderUsers();



};







document

.getElementById("prev")

.onclick = function(){



if(currentPage>1){



currentPage--;



renderUsers();



}



};









// -----------------------------------------------------
// SORT
// -----------------------------------------------------



document

.getElementById("sort")

.addEventListener(

"change",

function(){



if(this.value==="az"){



currentUsers.sort(

(a,b)=>

a.username.localeCompare(

b.username

)

);



}





if(this.value==="za"){



currentUsers.sort(

(a,b)=>

b.username.localeCompare(

a.username

)

);



}





renderUsers();



}

);

<script>


// =====================================================
// USER DETAIL

function openUser(username){



let user =

currentUsers.find(

u =>

u.username === username

);




if(!user)

return;







document

.getElementById(

"dUsername"

)

.innerHTML =


"<b>Kullanıcı:</b> @" +

user.username;








document

.getElementById(

"dStatus"

)

.innerHTML =


"<b>Durum:</b> " +

(

user.status

||

"Bilinmiyor"

);








document

.getElementById(

"dCategory"

)

.innerHTML =


"<b>Kategori:</b> " +

currentCategory;








document

.getElementById(

"dSource"

)

.innerHTML =


"<b>Kaynak:</b> " +

(

user.source

||

"Bulunamadı"

);








let date =

"Veri yok";






if(user.timestamp){



date =

new Date(

user.timestamp * 1000

)

.toLocaleString();



}








document

.getElementById(

"dDate"

)

.innerHTML =


"<b>Tarih:</b> " +

date;








document

.getElementById(

"dInstagram"

)

.href =

user.url;









document

.getElementById(

"userModal"

)

.style.display =

"flex";



}








function closeUser(){



document

.getElementById(

"userModal"

)

.style.display =

"none";



}









// -----------------------------------------------------
// ESC CLOSE
// -----------------------------------------------------

"keydown",

function(e){

}



}

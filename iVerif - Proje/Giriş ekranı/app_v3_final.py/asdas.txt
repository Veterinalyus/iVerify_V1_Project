<script>





// =====================================================
// DATA VARIABLES
// =====================================================



const API_URL = "http://127.0.0.1:5000";



let analysisResult = null;



let selectedCategory = "mutual";



let userList = [];



let currentPage = 1;



let pageLimit = 50;









// =====================================================
// ZIP UPLOAD
// =====================================================



document

.getElementById("zipInput")

.addEventListener(

"change",

function(){



if(this.files.length){


uploadArchive(

this.files[0]

);


}



}

);










async function uploadArchive(file){



let formData = new FormData();



formData.append(

"file",

file

);






document

.getElementById("loadingBox")

.style.display="block";






try{



const response = await fetch(

API_URL + "/api/analyze",

{

method:"POST",

body:formData

}

);






const result = await response.json();







if(result.success){



analysisResult=result.data;



loadDashboard();



}

else{


alert(result.error);


}



}

catch(error){



alert(

"Python motoruna bağlanılamadı"

);



}





document

.getElementById("loadingBox")

.style.display="none";



}









function loadDashboard(){



document

.getElementById("dashboard")

.classList

.remove("hidden");





document

.getElementById("analysisPanel")

.style.display="block";







let stats=analysisResult.stats;





document

.getElementById("followersCount")

.innerText=

stats.followers || 0;






document

.getElementById("followingCount")

.innerText=

stats.following || 0;






document

.getElementById("mutualCount")

.innerText=

stats.mutual || 0;






document

.getElementById("notFollowCount")

.innerText=

stats.not_following_back || 0;






document

.getElementById("pendingCount")

.innerText=

stats.pending || 0;






changeCategory(

"mutual"

);



}








// =====================================================
// CATEGORY
// =====================================================



document

.querySelectorAll(".category-btn")

.forEach(button=>{



button.onclick=function(){



document

.querySelectorAll(".category-btn")

.forEach(btn=>

btn.classList.remove(

"active"

)

);






this.classList.add(

"active"

);






changeCategory(

this.dataset.category

);



};



});









function changeCategory(category){



selectedCategory=category;



currentPage=1;




userList=

analysisResult.users[category]

||

[];




renderUsers();



}








// =====================================================
// USER RENDER
// =====================================================



function renderUsers(){



let container=document

.getElementById("usersContainer");



container.innerHTML="";





let search=document

.getElementById("searchInput")

.value

.toLowerCase();





let filtered=userList.filter(user=>



user.username

&&

user.username

.toLowerCase()

.includes(search)



);






let start=(currentPage-1)*pageLimit;


let end=start+pageLimit;






filtered

.slice(start,end)

.forEach(user=>{



let row=document.createElement(

"div"

);



row.className="user-row";





row.innerHTML=`

<div

class="user-name"

onclick="openUser('${user.username}')">

@${user.username}

</div>



<a

class="profile-btn"

href="${user.url || '#'}"

target="_blank">

Profil

</a>

`;





container.appendChild(row);



});







document

.getElementById("pageInfo")

.innerText=currentPage;



}

<script>





// =====================================================
// USER DETAIL
// =====================================================



function openUser(username){



let user=userList.find(

item => item.username===username

);





if(!user)

return;







document

.getElementById("detailUsername")

.innerHTML=

"<b>Kullanıcı:</b> @"+user.username;






document

.getElementById("detailCategory")

.innerHTML=

"<b>Kategori:</b> "+selectedCategory;






document

.getElementById("detailSource")

.innerHTML=

"<b>Kaynak:</b> "+

(

user.source

||

"JSON"

);






let date="Yok";





if(user.timestamp){



date=new Date(

user.timestamp*1000

)

.toLocaleString();



}







document

.getElementById("detailDate")

.innerHTML=

"<b>Tarih:</b> "+date;






document

.getElementById("detailProfile")

.href=

user.url || "#";






document

.getElementById("userModal")

.style.display="flex";



}









function closeUser(){



document

.getElementById("userModal")

.style.display="none";



}









// =====================================================
// SEARCH
// =====================================================



document

.getElementById("searchInput")

.addEventListener(

"input",

function(){



currentPage=1;



renderUsers();



}

);









// =====================================================
// SORT
// =====================================================



document

.getElementById("sortSelect")

.addEventListener(

"change",

function(){



if(this.value==="az"){



userList.sort(

(a,b)=>

a.username.localeCompare(

b.username

)

);



}





if(this.value==="za"){



userList.sort(

(a,b)=>

b.username.localeCompare(

a.username

)

);



}





renderUsers();



}

);









// =====================================================
// PAGINATION
// =====================================================



document

.getElementById("nextPage")

.onclick=function(){



let max=Math.ceil(

userList.length/pageLimit

);






if(currentPage<max){



currentPage++;



renderUsers();



}



};








document

.getElementById("previousPage")

.onclick=function(){



if(currentPage>1){



currentPage--;



renderUsers();



}



};









// =====================================================
// REPORT
// =====================================================



function showReport(){



if(!analysisResult.report)

return;






document

.getElementById("reportArea")

.style.display="block";







document

.getElementById("reportText")

.innerText=

JSON.stringify(

analysisResult.report,

null,

2

);



}









// ESC CLOSE



document.addEventListener(

"keydown",

function(event){



if(event.key==="Escape"){



closeUser();



}



}

);

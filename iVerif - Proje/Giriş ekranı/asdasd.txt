let currentData = null;
let archiveListCache = [];


function escapeHtml(str) {

    if (!str) return '';

    return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');

}


async function loadArchives() {

    try {

        const res = await fetch(
            'http://127.0.0.1:5000/api/arsivler'
        );

        const json = await res.json();

        if (
            json.success &&
            json.data &&
            json.data.length > 0
        ) {

            archiveListCache = json.data;

            let html = `
                <table class="archive-table">
                    <thead>
                        <tr>
                            <th>Dosya Adı</th>
                            <th>Tarih</th>
                            <th>Takipçi</th>
                            <th>Takip Edilen</th>
                            <th>İşlem</th>
                        </tr>
                    </thead>
                    <tbody>
            `;

            archiveListCache.forEach(arch => {

                const date =
                    new Date(
                        arch.created_at * 1000
                    ).toLocaleString();

                html += `
                    <tr>
                        <td>
                            ${escapeHtml(
                                arch.original_filename
                            )}
                        </td>

                        <td>
                            ${escapeHtml(date)}
                        </td>

                        <td>
                            ${arch.followers_count}
                        </td>

                        <td>
                            ${arch.following_count}
                        </td>

                        <td>
                            <button
                                class="tab-btn"
                                onclick="compareWithPrevious('${arch.id}')"
                            >
                                Geçmişle Karşılaştır
                            </button>
                        </td>
                    </tr>
                `;

            });

            html += `
                    </tbody>
                </table>
            `;

            document.getElementById(
                'archive-list-container'
            ).innerHTML = html;

        } else {

            document.getElementById(
                'archive-list-container'
            ).innerHTML = `
                <p
                    style="
                        color: var(--text-muted);
                        font-size: 0.9rem;
                    "
                >
                    Kayıtlı arşiv bulunamadı.
                </p>
            `;

        }

    } catch (e) {

        console.error(
            "Arşivler yüklenemedi",
            e
        );

        document.getElementById(
            'archive-list-container'
        ).innerHTML = `
            <p
                style="
                    color: var(--danger);
                    font-size: 0.9rem;
                "
            >
                Sunucu bağlantısı kurulamadı.
            </p>
        `;

    }

}


function handleFileSelect(event) {

    const file =
        event.target.files[0];

    if (file) {
        uploadArchive(file);
    }

}


async function uploadArchive(file) {

    const formData =
        new FormData();

    formData.append(
        "instagram_archive",
        file
    );

    document.getElementById(
        'upload-card'
    ).classList.add('hidden');

    document.getElementById(
        'loading-card'
    ).classList.remove('hidden');

    try {

        const response =
            await fetch(
                'http://127.0.0.1:5000/api/arsivle',
                {
                    method: 'POST',
                    body: formData
                }
            );

        const result =
            await response.json();

        if (result.success) {

            currentData =
                result.data;

            showResults(
                currentData
            );

            loadArchives();

        } else {

            alert(
                "Hata: " +
                (
                    result.error ||
                    "Bilinmeyen hata"
                )
            );

            resetUI();

        }

    } catch (err) {

        alert(
            "Sunucu bağlantısı kurulamadı."
        );

        resetUI();

    } finally {

        document.getElementById(
            'loading-card'
        ).classList.add('hidden');

    }

}


function showResults(data) {

    document.getElementById(
        'stat-followers'
    ).innerText =
        data.stats.followers_count;

    document.getElementById(
        'stat-following'
    ).innerText =
        data.stats.following_count;

    document.getElementById(
        'stat-mutual'
    ).innerText =
        data.stats.mutual_count;

    document.getElementById(
        'stat-not-back'
    ).innerText =
        data.stats.not_following_back_count;

    document.getElementById(
        'stat-fans'
    ).innerText =
        data.stats.fans_count;

    switchTab(
        'not_following_back'
    );

    document.getElementById(
        'result-card'
    ).classList.remove('hidden');

}


function switchTab(
    tabName,
    btnElement
) {

    document
        .querySelectorAll('.tab-btn')
        .forEach(
            btn =>
                btn.classList.remove(
                    'active'
                )
        );

    if (btnElement) {

        btnElement.classList.add(
            'active'
        );

    } else {

        const buttons =
            document.querySelectorAll(
                '.tabs .tab-btn'
            );

        const indexMap = {
            'not_following_back': 0,
            'fans': 1,
            'mutual': 2,
            'followers': 3,
            'following': 4
        };

        if (
            buttons[
                indexMap[tabName]
            ]
        ) {

            buttons[
                indexMap[tabName]
            ].classList.add(
                'active'
            );

        }

    }

    const container =
        document.getElementById(
            'user-list-container'
        );

    container.innerHTML = '';

    if (
        !currentData ||
        !currentData.results
    ) {
        return;
    }

    const list =
        currentData.results[
            tabName
        ] || [];

    if (list.length === 0) {

        container.innerHTML =
            '<div class="user-item">Kayıt bulunamadı.</div>';

        return;
    }

    list.forEach(user => {

        const div =
            document.createElement(
                'div'
            );

        div.className =
            'user-item';

        div.innerHTML = `
            <span>
                @${escapeHtml(
                    user.username
                )}
            </span>

            <span
                style="
                    color:var(--text-muted);
                    font-size:0.8rem;
                "
            >
                ${escapeHtml(
                    user.source_file || ''
                )}
            </span>
        `;

        container.appendChild(
            div
        );

    });

}


function resetUI() {

    document.getElementById(
        'result-card'
    ).classList.add('hidden');

    document.getElementById(
        'upload-card'
    ).classList.remove('hidden');

    document.getElementById(
        'file-input'
    ).value = '';

    currentData = null;

}


async function compareWithPrevious(
    currentId
) {

    try {

        const currentIndex =
            archiveListCache.findIndex(
                a => a.id === currentId
            );

        if (currentIndex <= 0) {

            alert(
                "Bu arşivden daha eski bir kayıt bulunamadı."
            );

            return;
        }

        const prevArch =
            archiveListCache[
                currentIndex - 1
            ];

        const compRes =
            await fetch(
                `http://127.0.0.1:5000/api/arsiv/${prevArch.id}/karsilastir/${currentId}`
            );

        const compJson =
            await compRes.json();

        if (compJson.success) {

            const data =
                compJson.data;

            alert(
                `Arşiv Karşılaştırma Sonucu (${escapeHtml(prevArch.original_filename)} → ${escapeHtml(archiveListCache[currentIndex].original_filename)}):\n\n` +

                `• Yeni Takipçiler: ${data.followers.new.length}\n` +

                `• Kaybedilen Takipçiler: ${data.followers.lost.length}\n` +

                `• Yeni Takip Edilenler: ${data.following.new.length}\n` +

                `• Takipten Çıkarılanlar: ${data.following.removed.length}\n` +

                `• Yeni Karşılıklı: ${data.mutual.new.length}\n` +

                `• Karşılıklılığı Bitenler: ${data.mutual.lost.length}`
            );

        } else {

            alert(
                "Karşılaştırma hatası: " +
                (
                    compJson.error ||
                    "Bilinmeyen hata"
                )
            );

        }

    } catch (e) {

        alert(
            "Sunucu bağlantısı kurulamadı."
        );

    }

}


window.onload =
    loadArchives;
